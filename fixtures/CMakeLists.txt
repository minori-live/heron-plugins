cmake_minimum_required(VERSION 3.25)
project(heron-plugin-bundle-probes LANGUAGES CXX)
add_executable(vst3-probe vst3-probe.cpp)
target_compile_features(vst3-probe PRIVATE cxx_std_17)
target_include_directories(vst3-probe PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/vst3sdk)
add_executable(revox-probe revox-probe.cpp)
target_compile_features(revox-probe PRIVATE cxx_std_17)
target_include_directories(revox-probe PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/vst3sdk)
if(UNIX)
  target_link_libraries(vst3-probe PRIVATE ${CMAKE_DL_LIBS})
  target_link_libraries(revox-probe PRIVATE ${CMAKE_DL_LIBS})
endif()
if(APPLE)
  add_executable(au-probe au-probe.cpp)
  target_link_libraries(au-probe PRIVATE "-framework AudioToolbox")
endif()
